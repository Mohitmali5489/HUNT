-- ========================================== -- 1. TABLES SETUP -- ========================================== -- Global settings for the Admin to control the game CREATE TABLE IF NOT EXISTS game_settings ( id INT PRIMARY KEY DEFAULT 1, is_registration_open BOOLEAN DEFAULT TRUE, max_teams INT DEFAULT 20 ); -- Insert the default row for admin controls (only row ID 1 is used) INSERT INTO game_settings (id, is_registration_open, max_teams) VALUES (1, TRUE, 20) ON CONFLICT (id) DO NOTHING; -- Table to store created Teams CREATE TABLE IF NOT EXISTS teams ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, captain_id TEXT NOT NULL, is_locked BOOLEAN DEFAULT FALSE, created_at TIMESTAMPTZ DEFAULT NOW() ); -- Table to store Team Members (linked to Teams) CREATE TABLE IF NOT EXISTS team_members ( team_id UUID REFERENCES teams(id) ON DELETE CASCADE, student_id TEXT PRIMARY KEY, -- Primary Key ensures 1 student can only join 1 team globally student_name TEXT NOT NULL, joined_at TIMESTAMPTZ DEFAULT NOW() ); -- Table to store Solo Participants CREATE TABLE IF NOT EXISTS solo_registrations ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), student_id TEXT UNIQUE NOT NULL, -- Ensures a student can only register solo once name TEXT NOT NULL, created_at TIMESTAMPTZ DEFAULT NOW() ); -- ========================================== -- 2. BUSINESS LOGIC FUNCTIONS (RPC) -- ========================================== -- A. Create a Team CREATE OR REPLACE FUNCTION create_team(p_team_name TEXT, p_student_id TEXT, p_student_name TEXT) RETURNS UUID LANGUAGE plpgsql AS $$ DECLARE v_team_id UUID; v_is_open BOOLEAN; BEGIN -- Check if lobby is open SELECT is_registration_open INTO v_is_open FROM game_settings WHERE id = 1; IF NOT v_is_open THEN RAISE EXCEPTION 'Registration is closed.'; END IF; -- Insert team and get generated UUID INSERT INTO teams (name, captain_id) VALUES (p_team_name, p_student_id) RETURNING id INTO v_team_id; -- Insert creator as the first team member (Leader) INSERT INTO team_members (team_id, student_id, student_name) VALUES (v_team_id, p_student_id, p_student_name); RETURN v_team_id; END; $$; -- B. Join an Existing Team CREATE OR REPLACE FUNCTION join_team(p_team_id UUID, p_student_id TEXT, p_student_name TEXT) RETURNS VOID LANGUAGE plpgsql AS $$ DECLARE v_member_count INT; v_is_locked BOOLEAN; v_is_open BOOLEAN; BEGIN -- Check if lobby is open SELECT is_registration_open INTO v_is_open FROM game_settings WHERE id = 1; IF NOT v_is_open THEN RAISE EXCEPTION 'Registration is closed.'; END IF; -- Check if team is already locked SELECT is_locked INTO v_is_locked FROM teams WHERE id = p_team_id; IF v_is_locked THEN RAISE EXCEPTION 'Team is already locked and cannot accept new members.'; END IF; -- Check if team is full (max 5) SELECT COUNT(*) INTO v_member_count FROM team_members WHERE team_id = p_team_id; IF v_member_count >= 5 THEN RAISE EXCEPTION 'This team is already full.'; END IF; -- Insert new member INSERT INTO team_members (team_id, student_id, student_name) VALUES (p_team_id, p_student_id, p_student_name); END; $$; -- C. Lock a Team (Submit Registration) CREATE OR REPLACE FUNCTION lock_team(p_team_id UUID, p_captain_id TEXT) RETURNS VOID LANGUAGE plpgsql AS $$ DECLARE v_member_count INT; v_locked_count INT; v_max_teams INT; BEGIN -- Check if requester is actually the leader/captain IF NOT EXISTS (SELECT 1 FROM teams WHERE id = p_team_id AND captain_id = p_captain_id) THEN RAISE EXCEPTION 'Only the Team Leader can lock the team.'; END IF; -- Ensure team has exactly 5 members SELECT COUNT(*) INTO v_member_count FROM team_members WHERE team_id = p_team_id; IF v_member_count != 5 THEN RAISE EXCEPTION 'Team must have exactly 5 members to lock.'; END IF; -- Enforce global limit of locked teams SELECT max_teams INTO v_max_teams FROM game_settings WHERE id = 1; SELECT COUNT(*) INTO v_locked_count FROM teams WHERE is_locked = TRUE; IF v_locked_count >= v_max_teams THEN RAISE EXCEPTION 'Maximum number of locked teams has been reached. Event is full.'; END IF; -- Lock the team UPDATE teams SET is_locked = TRUE WHERE id = p_team_id; END; $$; -- D. Register as a Solo Participant CREATE OR REPLACE FUNCTION register_solo(p_student_id TEXT, p_name TEXT) RETURNS VOID LANGUAGE plpgsql AS $$ DECLARE v_is_open BOOLEAN; BEGIN -- 1. Check if global registration is open SELECT is_registration_open INTO v_is_open FROM game_settings WHERE id = 1; IF NOT v_is_open THEN RAISE EXCEPTION 'Registration is currently closed.'; END IF; -- 2. Check if student is ALREADY IN A TEAM IF EXISTS (SELECT 1 FROM team_members WHERE student_id = p_student_id) THEN RAISE EXCEPTION 'You are already registered in a Treasure Hunt team!'; END IF; -- 3. Check if student is already registered as a solo participant IF EXISTS (SELECT 1 FROM solo_registrations WHERE student_id = p_student_id) THEN RAISE EXCEPTION 'This Student ID is already registered for solo play.'; END IF; -- 4. Insert the record INSERT INTO solo_registrations (student_id, name) VALUES (p_student_id, p_name); END; $$;
